Arduino UNO

Download .ino
const int SWITCH_PIN = 8;

void setup() {
  Serial.begin(9600);
  pinMode(SWITCH_PIN, OUTPUT);
}

void loop() {
  digitalWrite(SWITCH_PIN, HIGH);   // gate at about 5 V: fully on
  Serial.println("Load ON");
  delay(3000);
  digitalWrite(SWITCH_PIN, LOW);    // gate at 0 V: off
  Serial.println("Load OFF");
  delay(3000);
}

const int SWITCH_PIN = 25;

void setup() {
  Serial.begin(115200);
  pinMode(SWITCH_PIN, OUTPUT);
}

void loop() {
  // gate at about 3.3 V: on for this small load
  digitalWrite(SWITCH_PIN, HIGH);
  Serial.println("Load ON");
  delay(3000);
  digitalWrite(SWITCH_PIN, LOW);
  Serial.println("Load OFF");
  delay(3000);
}
