const int BUTTON_PIN = 2;
const unsigned long DEBOUNCE_MS = 20;

int lastReading = HIGH;          // pull-up: HIGH means released
int stableState = HIGH;
unsigned long lastChange = 0;
int presses = 0;

void setup() {
  Serial.begin(9600);
  pinMode(BUTTON_PIN, INPUT);    // external 10 k pull-up fitted
  // Without the external resistor, use this line instead:
  // pinMode(BUTTON_PIN, INPUT_PULLUP);
}

void loop() {
  int reading = digitalRead(BUTTON_PIN);
  if (reading != lastReading) {          // the pin changed: restart the timer
    lastChange = millis();
    lastReading = reading;
  }
  if (millis() - lastChange > DEBOUNCE_MS && reading != stableState) {
    stableState = reading;               // stable for 20 ms: accept it
    if (stableState == LOW) {            // active-low: LOW means pressed
      presses++;
      Serial.print("Presses: ");
      Serial.println(presses);
    }
  }
}

// not GPIO 34-39: they have no pull-ups
const int BUTTON_PIN = 27;
const unsigned long DEBOUNCE_MS = 20;

int lastReading = HIGH;
int stableState = HIGH;
unsigned long lastChange = 0;
int presses = 0;

void setup() {
  Serial.begin(115200);
  pinMode(BUTTON_PIN, INPUT);           // external 10 k pull-up to 3V3 fitted
  // Without the external resistor, use this line instead:
  // pinMode(BUTTON_PIN, INPUT_PULLUP);
}

void loop() {
  int reading = digitalRead(BUTTON_PIN);
  if (reading != lastReading) {
    lastChange = millis();
    lastReading = reading;
  }
  if (millis() - lastChange > DEBOUNCE_MS && reading != stableState) {
    stableState = reading;
    if (stableState == LOW) {
      presses++;
      Serial.printf("Presses: %d\n", presses);
    }
  }
}

Experiment 2: count every edge, no debounce (replace loop() with this)

int last = HIGH;
void loop() {                      // same code on the UNO and the ESP32
  int r = digitalRead(BUTTON_PIN);
  if (last == HIGH && r == LOW) {  // every HIGH-to-LOW change counts
    presses++;
    Serial.println(presses);
  }
  last = r;
}
