const float VREF = 1.1;      // measure the AREF pin with your meter and put the real value here
const float R_SHUNT = 10.0;  // ohm

void setup() {
  Serial.begin(9600);
  analogReference(INTERNAL); // 1.1 V reference: about 1.07 mV per step
  for (int n = 0; n < 5; n++) analogRead(A0);   // discard readings while the reference settles
}

void loop() {
  long sum = 0;
  for (int n = 0; n < 16; n++) sum += analogRead(A0);   // average 16 readings
  float vShunt = (sum / 16.0) * VREF / 1024.0;          // volts
  float current = vShunt / R_SHUNT * 1000.0;            // mA
  Serial.print("Shunt: ");    Serial.print(vShunt * 1000.0, 1);
  Serial.print(" mV   Current: "); Serial.print(current, 2);
  Serial.println(" mA");
  delay(500);
}

ESP32

Download .ino
const int   PIN = 34;
const float R_SHUNT = 10.0;  // ohm

void setup() {
  Serial.begin(115200);
  analogSetPinAttenuation(PIN, ADC_0db);  // 0 dB: accurate from about 100 to 950 mV
}

void loop() {
  long sum = 0;
  for (int n = 0; n < 16; n++) sum += analogReadMilliVolts(PIN);   // average 16 readings
  float mvShunt = sum / 16.0;
  float current = mvShunt / R_SHUNT;      // mV / ohm = mA
  Serial.printf("Shunt: %.1f mV   Current: %.2f mA\n", mvShunt, current);
  delay(500);
}

#include <Wire.h>
#include <Adafruit_INA219.h>

Adafruit_INA219 ina219;          // default I2C address 0x40

void setup() {
  Serial.begin(115200);          // UNO users can keep 115200 or change to 9600
  if (!ina219.begin()) {
    Serial.println("INA219 not found: check SDA, SCL, VCC and GND");
    while (true) delay(10);
  }
  ina219.setCalibration_16V_400mA();   // best resolution for small currents
}

void loop() {
  Serial.print("Bus: ");      Serial.print(ina219.getBusVoltage_V(), 2);
  Serial.print(" V   Shunt: ");   Serial.print(ina219.getShuntVoltage_mV(), 2);
  Serial.print(" mV   Current: "); Serial.print(ina219.getCurrent_mA(), 1);
  Serial.print(" mA   Power: ");   Serial.print(ina219.getPower_mW(), 0);
  Serial.println(" mW");
  delay(500);
}
