const int OUT_PIN = 8;
const int ADC_PIN = A0;
const float VS = 5.0;            // measure your 5V pin and put the real value here
const float C_TANK = 100e-9;     // the tank capacitor fitted (F)
const bool PLOT = false;         // true: print only the detector voltage, for the Serial Plotter

float readDet() {
  long sum = 0;
  for (int n = 0; n < 16; n++) sum += analogRead(ADC_PIN);
  return sum / 16.0 * VS / 1024.0;
}

void sweep() {
  float best = 0, fBest = 0;
  for (float f = 800; f <= 16000; f *= 1.04) {
    tone(OUT_PIN, (unsigned int)f);
    delay(40);                               // let the detector settle (4 x 10 ms)
    float v = readDet();
    if (PLOT) { Serial.print("det:"); Serial.println(v, 3); }
    else { Serial.print(f, 0); Serial.print(" Hz: "); Serial.print(v, 3); Serial.println(" V"); }
    if (v > best) { best = v; fBest = f; }
  }
  noTone(OUT_PIN);
  float Lcalc = 1.0 / (sq(2 * PI * fBest) * C_TANK);
  Serial.print("Peak at about "); Serial.print(fBest, 0); Serial.print(" Hz ("); Serial.print(best, 3);
  Serial.print(" V): L = "); Serial.print(Lcalc * 1000, 2); Serial.println(" mH");
}

void setup() {
  Serial.begin(9600);
  sweep();
}

void loop() {
  if (Serial.available()) {
    while (Serial.available()) Serial.read();
    sweep();
  }
}

const int OUT_PIN = 25;
const int ADC_PIN = 34;
const float C_TANK = 100e-9;     // the tank capacitor fitted (F)
const bool PLOT = false;         // true: print only the detector voltage, for the Serial Plotter

float readDet() {
  long sum = 0;
  for (int n = 0; n < 16; n++) sum += analogReadMilliVolts(ADC_PIN);
  return sum / 16.0 / 1000.0;
}

void sweep() {
  float best = 0, fBest = 0;
  for (float f = 800; f <= 16000; f *= 1.04) {
    ledcWriteTone(OUT_PIN, (uint32_t)f);     // 50 % square wave at f
    delay(40);
    float v = readDet();
    if (PLOT) Serial.printf("det:%.3f\n", v);
    else Serial.printf("%.0f Hz: %.3f V\n", f, v);
    if (v > best) { best = v; fBest = f; }
  }
  ledcWriteTone(OUT_PIN, 0);                  // stop
  float Lcalc = 1.0 / (pow(2 * PI * fBest, 2) * C_TANK);
  Serial.printf("Peak at about %.0f Hz (%.3f V): L = %.2f mH\n", fBest, best, Lcalc * 1000);
}

void setup() {
  Serial.begin(115200);
  analogSetPinAttenuation(ADC_PIN, ADC_0db);  // 0 dB: about 0.1-0.95 V, best for small signals
  ledcAttach(OUT_PIN, 1000, 8);               // ESP32 Arduino core 3.x
  sweep();
}

void loop() {
  if (Serial.available()) {
    while (Serial.available()) Serial.read();
    sweep();
  }
}
